namespace = fh_manila_petitions

fh_manila_petitions.1 = {
	type = country_event
	placement = root
	title = fh_manila_petitions.1.t
	desc = fh_manila_petitions.1.d
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 1

	trigger = { fh_manila_petition_warehouses_active = yes }
	cancellation_trigger = { fh_manila_petition_warehouses_active = no }

	option = {
		name = fh_manila_petitions.1.a
		trigger = { gold_reserves >= fh_manila_petition_cash }
		show_as_unavailable = { always = yes }
		custom_tooltip = fh_manila_petitions.1.a_tt
		hidden_effect = {
			if = {
				limit = {
					fh_manila_petition_warehouses_active = yes
					gold_reserves >= fh_manila_petition_cash
				}
				fh_manila_resolve_petition = { PETITION = warehouses }
				fh_manila_pay_petition = yes
				scope:fh_manila_petition_taiwan = {
					add_modifier = { name = fh_manila_bonded_warehouses months = 12 }
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { trigger = { scaled_gold_reserves >= 0.1 } add = 35 }
			modifier = { trigger = { ig:ig_industrialists ?= { is_in_government = yes } } add = 25 }
			modifier = {
				trigger = {
					scope:fh_manila_petition_taiwan ?= {
						any_scope_building = { is_building_type = building_trade_center level >= 3 }
					}
				}
				add = 30
			}
		}
	}
	option = {
		name = fh_manila_petitions.1.b
		custom_tooltip = fh_manila_petitions.1.b_tt
		hidden_effect = {
			if = {
				limit = { fh_manila_petition_warehouses_active = yes }
				fh_manila_resolve_petition = { PETITION = warehouses }
				add_treasury = { value = fh_manila_petition_cash }
				ig:ig_industrialists ?= {
					add_modifier = { name = fh_manila_warehouse_concession_denied months = 12 }
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { trigger = { scaled_gold_reserves < 0.05 } add = 75 }
		}
	}
	option = {
		name = fh_manila_petitions.dismiss
		default_option = yes
		ai_chance = { base = 0 }
		hidden_effect = { fh_manila_clear_pending_petition = { PETITION = warehouses } }
	}
}
