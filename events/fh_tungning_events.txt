namespace = fh_tungning

fh_tungning.1 = {
	type = country_event
	placement = root

	title = fh_tungning.1.t
	desc = fh_tungning.1.d

	event_image = { video = "asia_farmers_market" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:CHI ?= this
		is_player = yes
		c:TWG ?= {
			is_country_alive = yes
			has_variable = fh_tungning_bootstrapped
			is_player = no
		}
	}

	option = {
		name = fh_tungning.1.a
		default_option = yes
	}

	option = {
		name = fh_tungning.1.b
		trigger = {
			is_player = yes
			c:TWG ?= {
				is_country_alive = yes
				has_variable = fh_tungning_bootstrapped
				is_player = no
			}
		}
		play_as = c:TWG
		c:TWG = { trigger_event = { id = fh_tungning.2 popup = yes } }
		ai_chance = { base = 0 }
	}
}
