namespace = fh_tungning

fh_tungning.1 = {
	type = country_event
	placement = root

	title = fh_tungning.1.t
	desc = fh_tungning.1.d

	event_image = { video = "asia_farmers_market" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:CHI ?= this
		is_player = yes
		c:TWG ?= {
			is_country_alive = yes
			has_variable = fh_tungning_bootstrapped
			is_player = no
		}
	}

	option = {
		name = fh_tungning.1.a
		default_option = yes
	}

	option = {
		name = fh_tungning.1.b
		trigger = {
			is_player = yes
			c:TWG ?= {
				is_country_alive = yes
				has_variable = fh_tungning_bootstrapped
				is_player = no
			}
		}
		play_as = c:TWG
		c:TWG = { trigger_event = { id = fh_tungning.2 popup = yes } }
		ai_chance = { base = 0 }
	}
}

fh_tungning.2 = {
	type = country_event
	placement = root

	title = fh_tungning.2.t
	desc = fh_tungning.2.d
	flavor = {
		triggered_desc = {
			trigger = {
				exists = scope:fh_xu_wenjing
				exists = scope:fh_luo_jingde
				exists = scope:fh_samuel_talavan
			}
			desc = fh_tungning.2.f
		}
	}

	event_image = { video = "asia_farmers_market" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TWG ?= this
		has_variable = fh_tungning_bootstrapped
	}

	immediate = {
		random_scope_character = {
			limit = { has_template = fh_xu_wenjing_intelligentsia is_character_alive = yes }
			save_scope_as = fh_xu_wenjing
		}
		random_scope_character = {
			limit = { has_template = fh_luo_jingde_devout is_character_alive = yes }
			save_scope_as = fh_luo_jingde
		}
		random_scope_character = {
			limit = { has_template = fh_samuel_talavan_agitator is_character_alive = yes }
			save_scope_as = fh_samuel_talavan
		}
	}

	option = {
		name = fh_tungning.2.a
		default_option = yes
	}
}
