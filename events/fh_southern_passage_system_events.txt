namespace = fh_southern_passage_system

# Let any same-day native monthly progress and peace effects land first.
fh_southern_passage_system.1 = {
	type = country_event
	hidden = yes
	trigger = { fh_southern_passage_open = yes }
	immediate = {
		if = {
			limit = { NOT = { has_variable = fh_southern_passage_held } }
			fh_southern_passage_mark_due = yes
			fh_southern_passage_queue_reconcile = yes
		}
	}
}

fh_southern_passage_system.2 = {
	type = country_event
	hidden = yes
	trigger = {
		fh_southern_passage_open = yes
		has_variable = fh_southern_passage_reconcile_queued
	}
	immediate = {
		remove_variable = fh_southern_passage_reconcile_queued
		fh_southern_passage_reconcile = yes
	}
}
