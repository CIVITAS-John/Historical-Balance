namespace = fh_constitution

fh_constitution.1 = {
	type = country_event
	placement = root
	title = fh_constitution.1.t
	desc = fh_constitution.1.d
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 1

	trigger = { fh_constitution_choose_tone = { ROUTE = 1 READY = fh_constitution_republic_ready } }
	cancellation_trigger = { NOT = { fh_constitution_choose_tone = { ROUTE = 1 READY = fh_constitution_republic_ready } } }

	option = {
		name = fh_constitution_compact
		ai_chance = {
			base = 100
			modifier = {
				trigger = {
					var:fh_constitution_ming <= 150
					var:fh_constitution_zheng <= 150
				}
				add = -100
			}
		}
		custom_tooltip = fh_constitution_compact_tt
		hidden_effect = { fh_constitution_confirm = { ROUTE = 1 READY = fh_constitution_republic_ready TONE = 1 } }
	}
	option = {
		name = fh_constitution_proclamation
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					var:fh_constitution_ming <= 150
					var:fh_constitution_zheng <= 150
				}
				add = 100
			}
		}
		custom_tooltip = fh_constitution_proclamation_tt
		hidden_effect = { fh_constitution_confirm = { ROUTE = 1 READY = fh_constitution_republic_ready TONE = 2 } }
	}
	option = {
		name = fh_constitution_defer
		default_option = yes
		ai_chance = { base = 0 }
		hidden_effect = {
			if = {
				limit = { var:fh_constitution_ratification ?= 1 }
				remove_variable = fh_constitution_ratification
			}
		}
	}
}

fh_constitution.11 = {
	type = country_event
	placement = root
	title = fh_constitution.11.t
	desc = fh_constitution.11.d
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 1
	trigger = { fh_constitution_petition_valid = { PETITION = 11 ELIGIBLE = fh_constitution_civic_electorate } }
	cancellation_trigger = { NOT = { fh_constitution_petition_valid = { PETITION = 11 ELIGIBLE = fh_constitution_civic_electorate } } }

	option = {
		name = fh_constitution_petition_back
		custom_tooltip = fh_constitution_petition_back_tt
		ai_chance = {
			base = 50
			modifier = {
				trigger = { fh_constitution_recommend_republic = yes }
				add = 30
			}
		}
		hidden_effect = {
			fh_constitution_answer_petition = {
				PETITION = 11
				ELIGIBLE = fh_constitution_civic_electorate
				MANDATE = fh_constitution_republic
				DELTA = 5
				IG = ig_intelligentsia
				APPROVAL = fh_constitution_petition_approval
			}
		}
	}
	option = {
		name = fh_constitution_petition_reject
		custom_tooltip = fh_constitution_petition_reject_tt
		ai_chance = {
			base = 50
			modifier = {
				trigger = { fh_constitution_recommend_republic = yes }
				add = -30
			}
		}
		hidden_effect = {
			fh_constitution_answer_petition = {
				PETITION = 11
				ELIGIBLE = fh_constitution_civic_electorate
				MANDATE = fh_constitution_republic
				DELTA = -5
				IG = ig_intelligentsia
				APPROVAL = fh_constitution_petition_disapproval
			}
		}
	}
}
