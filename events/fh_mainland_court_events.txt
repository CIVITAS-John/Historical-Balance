fh_mainland_court.14 = {
	type = country_event
	placement = root
	title = fh_mainland_court.14.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { fh_mainland_officers_mainland = yes }
				desc = fh_mainland_court.14.d_mainland
			}
			triggered_desc = {
				trigger = { fh_mainland_officers_southern = yes }
				desc = fh_mainland_court.14.d_southern
			}
			desc = fh_mainland_court.14.d_neutral
		}
	}
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 1

	trigger = {
		fh_mainland_court_petition_pending = { PETITION = 14 }
		fh_mainland_officer_petition_eligible = yes
	}
	cancellation_trigger = {
		NAND = {
			fh_mainland_court_petition_pending = { PETITION = 14 }
			fh_mainland_officer_petition_eligible = yes
		}
	}

	option = {
		name = fh_mainland_court.14.back
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					OR = {
						AND = {
							fh_mainland_officers_mainland = yes
							fh_mainland_ai_favors_memorials = yes
						}
						AND = {
							fh_mainland_officers_southern = yes
							fh_mainland_ai_favors_memorials = no
						}
					}
				}
				add = 2
			}
		}
		if = {
			limit = { fh_mainland_officers_mainland = yes }
			custom_tooltip = fh_mainland_petition_plus_tt
		}
		else_if = {
			limit = { fh_mainland_officers_southern = yes }
			custom_tooltip = fh_mainland_petition_minus_tt
		}
		custom_tooltip = fh_mainland_court.14.back_tt
		hidden_effect = {
			fh_mainland_answer_officers = {
				BACKED = yes
				MAINLAND_DELTA = 5
				SOUTHERN_DELTA = -5
			}
		}
	}

	option = {
		name = fh_mainland_court.14.refuse
		default_option = yes
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					OR = {
						AND = {
							fh_mainland_officers_southern = yes
							fh_mainland_ai_favors_memorials = yes
						}
						AND = {
							fh_mainland_officers_mainland = yes
							fh_mainland_ai_favors_memorials = no
						}
					}
				}
				add = 2
			}
		}
		if = {
			limit = { fh_mainland_officers_mainland = yes }
			custom_tooltip = fh_mainland_petition_minus_tt
		}
		else_if = {
			limit = { fh_mainland_officers_southern = yes }
			custom_tooltip = fh_mainland_petition_plus_tt
		}
		custom_tooltip = fh_mainland_court.14.refuse_tt
		hidden_effect = {
			fh_mainland_answer_officers = {
				BACKED = no
				MAINLAND_DELTA = -5
				SOUTHERN_DELTA = 5
			}
		}
	}
}
