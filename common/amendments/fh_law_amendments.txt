	parent = law_fh_mandate_of_heaven
	allowed_laws = { law_fh_mandate_of_heaven }

	modifier = {
		state_standard_of_living_add = -0.5
		state_manchu_standard_of_living_add = 1.5

		country_acceptance_shared_heritage_trait_add = -10
		country_acceptance_shared_heritage_trait_group_add = -10
		country_acceptance_no_shared_heritage_trait_add = -10
		country_manchu_cultural_acceptance_add = 10

		state_radicals_and_loyalists_from_sol_change_mult = 0.25
		state_conscription_rate_mult = -0.50

	possible = {
		c:CHI ?= this
		NOT = { has_variable = fh_banner_precedence_removed }

	can_repeal = { legitimacy >= legitimacy_to_repeal_amendment }
}
