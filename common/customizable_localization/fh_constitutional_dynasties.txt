fh_constitution_ming_rites = {
	type = country
	random_valid = no

	text = {
		trigger = {
			var:fh_court_faith_settled_religion ?= rel:confucian
			country_has_state_religion = rel:confucian
		}
		localization_key = fh_constitution_ming_rites_confucian
	}
	text = {
		trigger = {
			var:fh_court_faith_settled_religion ?= rel:mahayana
			country_has_state_religion = rel:mahayana
		}
		localization_key = fh_constitution_ming_rites_mahayana
	}
	text = {
		trigger = {
			OR = {
				AND = {
					var:fh_court_faith_settled_religion ?= rel:catholic
					country_has_state_religion = rel:catholic
				}
				AND = {
					var:fh_court_faith_settled_religion ?= rel:protestant
					country_has_state_religion = rel:protestant
				}
			}
		}
		localization_key = fh_constitution_ming_rites_christian
	}
	text = { localization_key = fh_constitution_ming_rites_deferred }
}
