fh_manila_provinces_recorded = {
	value = 0
	every_in_list = {
		variable = fh_manila_starting_provinces
		add = 1
	}
}

fh_manila_provinces_held = {
	value = 0
	every_in_list = {
		variable = fh_manila_starting_provinces
		if = { limit = { owner ?= prev } add = 1 }
	}
}

# A positive balance is a strict majority of the recorded province set.
fh_manila_recorded_balance = {
	value = 0
	every_in_list = {
		variable = fh_manila_starting_provinces
		if = {
			limit = { owner ?= prev }
			add = 1
		}
		else = { subtract = 1 }
	}
}
