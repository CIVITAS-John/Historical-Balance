# An unavailable mandate scores zero. Weight the current laws first,
# then government and the holdings that would help finish the settlement.
fh_constitution_ai_republic = {
	value = 0
	if = {
		limit = { fh_constitution_republic_ready = yes }
		add = 1
		if = { limit = { fh_constitution_republic_law = yes } add = 8 }
		if = { limit = { fh_constitution_republic_franchise = yes } add = 4 }
		if = { limit = { fh_constitution_civic_sponsor = yes } add = 2 }
		if = { limit = { fh_constitution_recommend_republic = yes } add = 0.5 }
	}
}
