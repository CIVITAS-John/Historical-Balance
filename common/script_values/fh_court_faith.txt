# Other religions count in the population denominator but back neither side.
fh_court_faith_taiwan_demographic_drift = {
	value = 0
	every_scope_state = {
		limit = {
			state_region = s:STATE_FORMOSA
			state_population > 0
		}
		add = "this.religion_percent_state(rel:catholic)"
		add = "this.religion_percent_state(rel:protestant)"
		subtract = "this.religion_percent_state(rel:confucian)"
		subtract = "this.religion_percent_state(rel:mahayana)"
	}
}

# A stable order used only when the eligible sponsors have exactly equal clout.
fh_court_faith_ig_tiebreak = {
	value = 0
	if = { limit = { is_interest_group_type = ig_landowners } add = 8 }
	else_if = { limit = { is_interest_group_type = ig_devout } add = 7 }
	else_if = { limit = { is_interest_group_type = ig_intelligentsia } add = 6 }
	else_if = { limit = { is_interest_group_type = ig_industrialists } add = 5 }
	else_if = { limit = { is_interest_group_type = ig_armed_forces } add = 4 }
	else_if = { limit = { is_interest_group_type = ig_petty_bourgeoisie } add = 3 }
	else_if = { limit = { is_interest_group_type = ig_rural_folk } add = 2 }
	else_if = { limit = { is_interest_group_type = ig_trade_unions } add = 1 }
}
