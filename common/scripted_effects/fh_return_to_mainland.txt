fh_clear_legacy_mainland_registration = {
	remove_variable = fh_mainland_registration_pending
	remove_variable = fh_fujian_readiness
	remove_variable = fh_guangdong_readiness
	remove_variable = fh_zhejiang_readiness
}

fh_finish_mainland_registration = {
	hidden_effect = { set_variable = fh_return_to_mainland_completed }
	active_law:lawgroup_citizenship ?= { fh_retire_tungning_register_levies = yes }
	# An older save can register before its first law migration pulse.
	if = {
		limit = { has_modifier = fh_register_conscription }
		remove_modifier = fh_register_conscription
	}
	hidden_effect = { fh_clear_legacy_mainland_registration = yes }
}
