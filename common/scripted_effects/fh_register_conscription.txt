# Country scope. Run on activation and when the mainland charter ends the levy.
fh_sync_register_conscription = {
	if = {
		limit = {
			has_law = law_type:law_fh_port_and_village_registers
			NOT = { has_variable = fh_return_to_mainland_completed }
		}
		if = {
			limit = { NOT = { has_modifier = fh_register_conscription } }
			add_modifier = { name = fh_register_conscription }
		}
	}
	else_if = {
		limit = { has_modifier = fh_register_conscription }
		remove_modifier = fh_register_conscription
	}
}
