fh_mainland_start_mobilization = {
	if = {
		limit = { NOT = { has_variable = fh_mainland_mobilization_started } }
		set_variable = fh_mainland_mobilization_started
		set_variable = fh_mainland_mobilizing
		add_modifier = { name = fh_mainland_mobilization years = 3 }
		ig:ig_landowners ?= {
			add_modifier = { name = fh_mainland_mobilization_support years = 3 }
		}
		ig:ig_industrialists ?= {
			add_modifier = { name = fh_mainland_mobilization_protest years = 3 }
		}
		if = {
			limit = { fh_mainland_officers_mainland = yes }
			ig:ig_armed_forces ?= {
				add_modifier = { name = fh_mainland_mobilization_support years = 3 }
			}
		}
		else_if = {
			limit = { fh_mainland_officers_southern = yes }
			ig:ig_armed_forces ?= {
				add_modifier = { name = fh_mainland_mobilization_officer_protest years = 3 }
			}
		}
		if = {
			limit = { is_ai = yes }
			set_strategy = fh_mainland_mobilization_strategy
		}
		if = {
			limit = {
				has_variable = fh_mainland_from_taiping
				fh_mainland_taiping_active = yes
			}
			set_variable = fh_mainland_mobilization_taiping
		}
		fh_mainland_find_active_war = yes
		if = {
			# A peace transfer can fire the ownership check after war resolution.
			limit = { has_variable = fh_mainland_qing_war_ended_today }
			set_variable = fh_mainland_mobilization_followed_through
			if = {
				limit = { has_variable = fh_mainland_qing_war_defeat_today }
				fh_mainland_record_war_defeat = yes
			}
			fh_mainland_finish_mobilization = yes
		}
		else = { trigger_event = { id = fh_mainland_mobilization.1 years = 3 } }
	}
}

fh_mainland_find_active_war = {
	if = {
		limit = {
			has_variable = fh_mainland_mobilizing
			NOT = { has_variable = fh_mainland_first_war }
		}
		ordered_scope_war = {
			limit = {
				diplomatic_play ?= { fh_mainland_play_has_opposing_qing = yes }
			}
			order_by = "war_duration_months"
			position = 0
			check_range_bounds = no
			diplomatic_play = { save_temporary_scope_as = fh_mainland_active_war }
		}
		if = {
			limit = { exists = scope:fh_mainland_active_war }
			fh_mainland_record_war = { PLAY = scope:fh_mainland_active_war }
		}
	}
}

fh_mainland_record_war = {
	if = {
		limit = {
			has_variable = fh_mainland_mobilizing
			NOT = { has_variable = fh_mainland_first_war }
		}
		set_variable = { name = fh_mainland_first_war value = $PLAY$ }
		set_variable = fh_mainland_mobilization_followed_through
	}
}

fh_mainland_finish_mobilization = {
	if = {
		limit = { has_variable = fh_mainland_mobilizing }
		remove_variable = fh_mainland_mobilizing
		set_variable = fh_constitutional_debate_unlocked
		if = {
			limit = { has_modifier = fh_mainland_mobilization }
			remove_modifier = fh_mainland_mobilization
		}
		fh_mainland_remove_mobilization_approval = { IG = ig:ig_landowners NAME = fh_mainland_mobilization_support }
		fh_mainland_remove_mobilization_approval = { IG = ig:ig_industrialists NAME = fh_mainland_mobilization_protest }
		fh_mainland_remove_mobilization_approval = { IG = ig:ig_armed_forces NAME = fh_mainland_mobilization_support }
		fh_mainland_remove_mobilization_approval = { IG = ig:ig_armed_forces NAME = fh_mainland_mobilization_officer_protest }
		if = {
			limit = { has_strategy = fh_mainland_mobilization_strategy }
			set_strategy = ai_strategy_industrial_expansion
		}
		if = {
			limit = {
				NOT = { has_variable = fh_mainland_mobilization_followed_through }
				NAND = {
					has_variable = fh_mainland_mobilization_taiping
					has_variable = fh_taiping_aid_delivered
				}
			}
			add_modifier = { name = fh_mainland_war_never_fought years = 1 }
			if = {
				limit = { fh_mainland_officers_mainland = yes }
				ig:ig_armed_forces ?= {
					add_modifier = { name = fh_mainland_officer_disappointment years = 1 }
				}
			}
			trigger_event = { id = fh_mainland_mobilization.2 popup = yes }
		}
		fh_constitution_try_open = yes
	}
}

fh_mainland_remove_mobilization_approval = {
	$IG$ ?= {
		if = {
			limit = { has_modifier = $NAME$ }
			remove_modifier = $NAME$
		}
	}
}

fh_mainland_record_war_defeat = {
	if = {
		limit = { NOT = { has_variable = fh_mainland_first_war_defeat } }
		set_variable = fh_mainland_first_war_defeat
		add_modifier = { name = fh_mainland_war_defeat years = 1 }
		trigger_event = { id = fh_mainland_mobilization.3 popup = yes }
	}
}

fh_mainland_open_claims_audience = {
	if = {
		limit = { fh_mainland_claims_decision_visible = yes }
		ordered_country = {
			limit = { fh_mainland_claim_target_ready = yes }
			order_by = army_power_projection
			position = 0
			check_range_bounds = no
			save_scope_as = fh_claims_target
		}
		if = {
			limit = { exists = scope:fh_claims_target }
			if = {
				limit = { has_variable = fh_mainland_claims_serial }
				change_variable = { name = fh_mainland_claims_serial add = 1 }
			}
			else = { set_variable = { name = fh_mainland_claims_serial value = 1 } }
			save_scope_value_as = { name = fh_claims_serial value = var:fh_mainland_claims_serial }
			set_variable = { name = fh_mainland_claims_audience days = 30 }
			trigger_event = { id = fh_mainland_mobilization.4 popup = yes }
		}
	}
}

fh_mainland_launch_claims_play = {
	if = {
		limit = {
			fh_mainland_claims_audience_valid = yes
			scope:fh_claims_target ?= { fh_mainland_claim_target_ready = yes }
		}
		remove_variable = fh_mainland_claims_audience
		scope:fh_claims_target = {
			random_scope_state = {
				limit = { fh_mainland_claim_target_state = yes }
				save_temporary_scope_as = fh_claims_first
			}
		}
		create_diplomatic_play = {
			name = dp_fh_mainland_claims
			type = dp_fh_mainland_claims
			target_country = scope:fh_claims_target
			target_state = scope:fh_claims_first
		}
		random_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_fh_mainland_claims
				initiator_is = global_var:fh_tungning_founder
				target_is = scope:fh_claims_target
			}
			save_temporary_scope_as = fh_claims_play
			scope:fh_claims_target = {
				every_scope_state = {
					limit = {
						NOT = { this = scope:fh_claims_first }
						fh_mainland_claim_target_state = yes
					}
					scope:fh_claims_play = {
						add_war_goal = {
							holder = global_var:fh_tungning_founder
							type = fh_mainland_return_state
							target_state = prev
							primary_demand = yes
						}
					}
				}
			}
		}
	}
}
