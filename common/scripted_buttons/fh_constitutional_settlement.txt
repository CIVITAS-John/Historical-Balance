fh_constitution_civic_congress = {
	name = fh_constitution_civic_congress
	desc = fh_constitution_civic_congress_desc
	visible = { fh_constitution_debate_open = yes }
	possible = {
		fh_constitution_endorsement_open = yes
		fh_constitution_civic_sponsor = yes
	}
	effect = {
		custom_tooltip = fh_constitution_endorsement_effect_tt
		hidden_effect = {
			fh_constitution_endorse = {
				ROUTE = 1
				SPONSOR = fh_constitution_civic_sponsor
				ELIGIBLE = always
				MANDATE = fh_constitution_republic
				RIVAL_1 = fh_constitution_ming
				RIVAL_2 = fh_constitution_zheng
				FIRST = ig_intelligentsia
				SECOND = ig_trade_unions
				OPPONENT = ig_landowners
			}
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				fh_constitution_ai_endorsement_open = yes
				OR = { fh_constitution_republic_law = yes fh_constitution_civic_electorate = yes }
			}
			add = 10
		}
	}
}
