fh_register_mainland_administration = {
	name = fh_register_mainland_administration
	desc = fh_register_mainland_administration_desc
	visible = { always = yes }
	possible = {
		NOT = { has_variable = fh_mainland_registration_pending }
		custom_tooltip = {
			text = fh_mainland_registration_ready_tt
			OR = {
				fh_mainland_region_can_register = { REGION = STATE_FUJIAN STREAK = fh_fujian_readiness }
				fh_mainland_region_can_register = { REGION = STATE_GUANGDONG STREAK = fh_guangdong_readiness }
				fh_mainland_region_can_register = { REGION = STATE_ZHEJIANG STREAK = fh_zhejiang_readiness }
			}
		}
	}
	ai_chance = { value = 100 }
	effect = {
		if = {
			limit = { fh_mainland_region_can_register = { REGION = STATE_FUJIAN STREAK = fh_fujian_readiness } }
			fh_select_mainland_registration_state = { REGION = STATE_FUJIAN }
		}
		else_if = {
			limit = { fh_mainland_region_can_register = { REGION = STATE_GUANGDONG STREAK = fh_guangdong_readiness } }
			fh_select_mainland_registration_state = { REGION = STATE_GUANGDONG }
		}
		else_if = {
			limit = { fh_mainland_region_can_register = { REGION = STATE_ZHEJIANG STREAK = fh_zhejiang_readiness } }
			fh_select_mainland_registration_state = { REGION = STATE_ZHEJIANG }
		}
		if = {
			limit = { exists = scope:fh_registration_state }
			set_variable = { name = fh_mainland_registration_pending days = 32 }
			trigger_event = { id = fh_mainland_registration.1 popup = yes }
		}
	}
}
