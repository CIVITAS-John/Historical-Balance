fh_southern_passage_quarterly_events = {
	random_events = {
		chance_to_happen = 10
		1 = fh_southern_passage_action.11
		1 = fh_southern_passage_action.12
	}
}

on_law_activated = {
	on_actions = { fh_southern_passage_check_tariff_law }
}

fh_southern_passage_check_tariff_law = {
	trigger = { owner ?= { has_modifier = fh_southern_passage_tariff_relief } }
	effect = { owner ?= { fh_southern_passage_refresh_tariff = yes } }
}

on_merge_markets = {
	on_actions = { fh_southern_passage_check_tariff_market }
}

on_create_market = {
	on_actions = { fh_southern_passage_check_tariff_market }
}

fh_southern_passage_check_tariff_market = {
	trigger = {
		global_var:fh_tungning_founder ?= {
			has_modifier = fh_southern_passage_tariff_relief
			NOT = { has_variable = fh_southern_passage_market_refresh_pending }
		}
	}
	effect = {
		# Market creation and merger may precede the country's market link update.
		global_var:fh_tungning_founder ?= {
			set_variable = fh_southern_passage_market_refresh_pending
			trigger_event = { id = fh_southern_passage_action.13 days = 1 }
		}
	}
}
