on_game_started = {
	on_actions = { fh_tungning_bookmark_setup }
}

fh_tungning_bookmark_setup = {
	trigger = { game_date = 1836.1.1 }

	effect = {
		c:DAI ?= {
			if = {
				limit = {
					is_country_alive = yes
					has_law = law_type:law_hereditary_bureaucrats
					any_primary_culture = { has_discrimination_trait = tradition_sinosphere }
				}
				activate_law = law_type:law_fh_standardized_examinations
			}
		}
		c:KOR ?= {
			if = {
				limit = {
					is_country_alive = yes
					has_law = law_type:law_appointed_bureaucrats
				}
				activate_law = law_type:law_fh_yangban_administration
			}
		}
		c:CHI ?= {
			if = {
				limit = { is_country_alive = yes }
				fh_tungning_setup = yes
				if = {
					limit = { NOT = { has_variable = fh_qing_laws_initialized } }
					set_variable = fh_qing_laws_initialized
					fh_qing_law_setup = yes
				}
			}
		}
		c:TWG ?= {
			if = {
				limit = {
					is_country_alive = yes
					has_variable = fh_tungning_bootstrapped
					NOT = { has_variable = fh_tungning_diplomacy_initialized }
				}
				set_variable = fh_tungning_diplomacy_initialized
				fh_tungning_diplomacy = yes
				trigger_event = { id = fh_tungning.3 days = 8 }
			}
		}
	}
}

on_game_started_after_lobby = {
	on_actions = { fh_tungning_opening }
}

fh_tungning_opening = {
	effect = {
		c:TWG ?= {
			if = {
				limit = {
					is_country_alive = yes
					has_variable = fh_tungning_bootstrapped
				}
				if = {
					limit = { is_player = yes }
					trigger_event = { id = fh_tungning.2 popup = yes }
				}
				else = {
					c:CHI ?= {
						if = {
							limit = {
								is_country_alive = yes
								is_player = yes
							}
							trigger_event = { id = fh_tungning.1 popup = yes }
						}
					}
				}
			}
		}
	}
}
