fh_je_mainland_foothold = {
	group = je_group_foreign_affairs
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	transferable = no
	can_revolution_inherit = no

	is_shown_when_inactive = {
		fh_is_tungning_founder = yes
		NOR = {
			has_variable = fh_return_to_mainland_completed
			has_variable = fh_mainland_claim_abandoned
		}
	}

	possible = { fh_mainland_registration_available = yes }

	progressbar = yes
	progress_desc = fh_je_mainland_foothold_progress
	current_value = fh_mainland_foothold_progress
	goal_add_value = {
		value = 50
		subtract = fh_mainland_foothold_progress
	}

	map_color = {
		if = {
			limit = {
				fh_mainland_coast_state = yes
				is_treaty_port = no
			}
			if = {
				limit = {
					owner ?= { fh_is_tungning_founder = yes }
					fh_mainland_state_unoccupied = yes
				}
				value = "russification_blue"
			}
			else = { value = "russification_amber" }
		}
	}

	map_tooltip = {
		if = {
			limit = { fh_mainland_coast_state = yes }
			if = {
				limit = { is_treaty_port = yes }
				custom_tooltip_no_bullet = fh_foothold_map_treaty_port
			}
			else_if = {
				limit = { NOT = { owner ?= { fh_is_tungning_founder = yes } } }
				custom_tooltip_no_bullet = fh_foothold_map_missing
			}
			else = {
				if = {
					limit = { fh_mainland_foothold_leader_state = yes }
					custom_tooltip_no_bullet = fh_foothold_map_leader
				}
				if = {
					limit = { fh_mainland_state_unoccupied = no }
					custom_tooltip_no_bullet = fh_foothold_map_occupied
				}
				else_if = {
					limit = { is_incorporated = yes }
					custom_tooltip_no_bullet = fh_foothold_map_incorporated
				}
				else = { custom_tooltip_no_bullet = fh_foothold_map_project }
				fh_mainland_integration_tooltip = yes
			}
		}
	}

	outcome = {
		header = fh_mainland_claim_abandoned_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = fh_je_mainland_foothold_abandoned_desc
		color = "blue_negative"
		trigger = { has_variable = fh_mainland_claim_abandoned }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = fh_je_mainland_foothold_completed_desc
		color = "gold_positive"
		trigger = {
			# An older save can already have recorded registration.
			trigger_if = {
				limit = { NOT = { has_variable = fh_return_to_mainland_completed } }
				fh_mainland_registration_available = yes
				custom_tooltip = {
					text = fh_mainland_registration_milestone
					OR = {
						fh_mainland_region_can_register = { REGION = STATE_FUJIAN }
						fh_mainland_region_can_register = { REGION = STATE_GUANGDONG }
						fh_mainland_region_can_register = { REGION = STATE_ZHEJIANG }
					}
				}
			}
		}
		effect = {
			if = {
				limit = { NOT = { has_variable = fh_return_to_mainland_completed } }
				fh_finish_mainland_registration = yes
			}
		}
	}
}
