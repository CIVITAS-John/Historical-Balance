	scripted_button = fh_constitution_invite_zhu
	scripted_button = fh_constitution_zheng_petition
	scripted_button = fh_constitution_ratify_republic
	scripted_button = fh_constitution_ratify_ming
	scripted_button = fh_constitution_ratify_zheng

	on_weekly_pulse = { effect = { fh_constitution_review = yes } }
	on_monthly_pulse = { effect = { fh_constitution_debate_monthly = yes } }

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { fh_constitution_ratification_valid = yes }
				desc = fh_constitution_choose_tone_status
			}
			triggered_desc = {
				trigger = { fh_constitution_recommend_zheng = yes }
				desc = fh_constitution_recommend_zheng_status
			}
			triggered_desc = {
				trigger = { fh_constitution_recommend_republic = yes }
				desc = fh_constitution_recommend_republic_status
			}
			triggered_desc = {
				trigger = { fh_constitution_recommend_ming = yes }
				desc = fh_constitution_recommend_ming_status
			}
			desc = fh_constitution_debate_status
		}
	}

	outcome = {
		header = fh_constitution_republic_ratified
		effect_header = je_standard_on_completion_header
		conclusion_desc = fh_constitution_republic_ratified_desc
		color = "gold_positive"
		trigger = { var:fh_constitution_route ?= 1 }
	}

	outcome = {
		header = fh_constitution_ming_ratified
		effect_header = je_standard_on_completion_header
		conclusion_desc = fh_constitution_ming_ratified_desc
		color = "gold_positive"
		trigger = { var:fh_constitution_route ?= 2 }
	}

	outcome = {
		header = fh_constitution_zheng_ratified
		effect_header = je_standard_on_completion_header
		conclusion_desc = fh_constitution_zheng_ratified_desc
		color = "gold_positive"
		trigger = { var:fh_constitution_route ?= 3 }
	}
}
