	market.owner ?= this

		has_law_or_variant = law_type:law_protectionism
	}

}

}

	fh_southern_passage_actions_open = yes
	NOT = { has_variable = fh_southern_passage_event_cooldown }
	$ELIGIBLE$ = yes
}

	fh_southern_passage_active = yes
	var:fh_southern_passage_pending_event ?= $EVENT$
	var:fh_southern_passage_rival ?= scope:fh_southern_passage_event_rival
	$ELIGIBLE$ = yes
}

	fh_southern_passage_pending_event_valid = { EVENT = $EVENT$ ELIGIBLE = $ELIGIBLE$ }
	NOT = { has_variable = fh_southern_passage_reconcile_queued }
}
