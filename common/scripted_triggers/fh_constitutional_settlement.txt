fh_constitution_claim_resolved = {
	fh_is_tungning_founder = yes
	has_variable = fh_constitutional_debate_unlocked
	OR = {
		has_variable = fh_mainland_claim_renounced
		AND = {
			has_variable = fh_mainland_claim_reasserted
			has_variable = fh_mainland_mobilization_started
		}
	}
	NOR = {
		has_variable = fh_strait_offer_pending
		has_variable = fh_strait_ratification_in_progress
		has_variable = fh_taiping_fork_pending
		has_variable = fh_mainland_mobilizing
	}
}
