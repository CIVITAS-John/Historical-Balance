fh_reunification_route_open = {
	fh_is_tungning_founder = yes
	NOR = {
		has_variable = fh_mainland_claim_abandoned
		has_variable = fh_taiping_fork_pending
	}
}

fh_reunification_provincial_access = {
	fh_reunification_route_open = yes
	has_variable = fh_reunification_opened
	is_subject = no
	OR = {
		fh_owns_mainland_foothold = { REGION = STATE_FUJIAN }
		fh_owns_mainland_foothold = { REGION = STATE_GUANGDONG }
		fh_owns_mainland_foothold = { REGION = STATE_ZHEJIANG }
	}
}

fh_reunification_can_negotiate = {
	fh_reunification_provincial_access = yes
	has_journal_entry = fh_je_reunification_of_china
	NOT = { has_variable = fh_reunification_proclaimed }
}

	has_journal_entry = fh_je_reunification_of_china

		fh_reunification_direct_region_secure = { REGION = STATE_GUANGDONG }
		fh_reunification_direct_region_secure = { REGION = STATE_ZHEJIANG }

	custom_tooltip = {

	}
