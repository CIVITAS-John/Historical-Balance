# Run in the country whose holdings are being checked. Foreign treaty-port
# enclaves are exempt; ordinary territory and subject holdings are not.
fh_owns_mainland_foothold = {
	save_temporary_scope_as = fh_foothold_owner
	s:$REGION$ = {
		any_scope_state = { owner = scope:fh_foothold_owner is_treaty_port = no }
		NOT = {
			any_scope_state = {
				is_treaty_port = no
				NOT = { owner = scope:fh_foothold_owner }
			}
		}
	}
}

fh_mainland_coast_state = {
	state_region = { fh_reunification_coastal_region = yes }
}

fh_mainland_state_unoccupied = {
	NOT = { any_province = { NOT = { controller ?= owner } } }
}

# Use after the direct ownership check. A foreign treaty port does not block
# registration or proclamation even if its own provinces are occupied.
fh_mainland_region_unoccupied = {
	s:$REGION$ = {
		NOT = {
			any_scope_state = {
				is_treaty_port = no
				fh_mainland_state_unoccupied = no
			}
		}
	}
}

fh_mainland_foothold_eligible = {
	fh_owns_mainland_foothold = { REGION = $REGION$ }
	fh_mainland_region_unoccupied = { REGION = $REGION$ }
}

fh_mainland_region_can_register = {
	fh_mainland_foothold_eligible = { REGION = $REGION$ }
	s:$REGION$ = {
		any_scope_state = {
			is_treaty_port = no
			OR = {
				is_incorporated = yes
				incorporation_progress >= 0.50
			}
		}
	}
}

fh_mainland_foothold_leader_state = {
	OR = {
		AND = {
			state_region = s:STATE_FUJIAN
			owner ?= { fh_mainland_foothold_leader = 1 }
		}
		AND = {
			state_region = s:STATE_GUANGDONG
			owner ?= { fh_mainland_foothold_leader = 2 }
		}
		AND = {
			state_region = s:STATE_ZHEJIANG
			owner ?= { fh_mainland_foothold_leader = 3 }
		}
	}
}

fh_mainland_registration_available = {
	fh_is_tungning_founder = yes
	has_variable = fh_mainland_foothold_opened
	NOR = {
		has_variable = fh_return_to_mainland_completed
		has_variable = fh_mainland_claim_abandoned
		has_variable = fh_taiping_fork_pending
	}
}
