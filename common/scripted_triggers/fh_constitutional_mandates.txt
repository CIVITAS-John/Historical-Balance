# State scope: an ordinary holding counts once only if no other country holds
# ordinary land in its region. The owner's subjects are other owners here.
fh_constitution_owned_inland_state = {
	is_treaty_port = no
	save_temporary_scope_as = fh_constitution_inland_holding
	state_region = {
		fh_constitution_inland_core_region = yes
		NOT = {
			any_scope_state = {
				is_treaty_port = no
				NOT = { owner = scope:fh_constitution_inland_holding.owner }
			}
		}
	}
}

# The faith journal assigns the Devout profiles. Older campaigns already
# recorded the adopted faith, or still have their opening Catholic profile.

		AND = {

			var:fh_court_faith_settled_religion ?= rel:confucian
		}
	}
}

	}
}

# JE03 contract: set the country success flag and the flags for original grants
# still held at victory. Leave those flags fixed; the check reads current holders.
fh_constitution_southern_passage = {
	has_variable = fh_southern_passage_secured
	OR = {
		fh_constitution_keeps_passage_subject = { TAG = STG GRANT = fh_southern_passage_secured_stg }
		fh_constitution_keeps_passage_subject = { TAG = SEL GRANT = fh_southern_passage_secured_sel }
		fh_constitution_keeps_passage_subject = { TAG = BLG GRANT = fh_southern_passage_secured_blg }
		fh_constitution_keeps_passage_province = { PROVINCE = x1C7417 GRANT = fh_southern_passage_secured_x1c7417 }
		fh_constitution_keeps_passage_province = { PROVINCE = x2A3D46 GRANT = fh_southern_passage_secured_x2a3d46 }
		fh_constitution_keeps_passage_province = { PROVINCE = xE02CB6 GRANT = fh_southern_passage_secured_xe02cb6 }
	}
}

fh_constitution_keeps_passage_subject = {
	has_variable = $GRANT$
	c:$TAG$ ?= {
		is_country_alive = yes
		overlord ?= prev
	}
}

fh_constitution_keeps_passage_province = {
	has_variable = $GRANT$
	p:$PROVINCE$ ?= { owner ?= prev }
}

fh_constitution_philippine_rule = {
	OR = {
		c:PHI ?= {
			is_country_alive = yes
			overlord ?= prev
			p:x0D8236 ?= { owner ?= c:PHI }
		}
		AND = {
			p:x0D8236 ?= { owner ?= prev }
			fh_constitution_manila_conquest_holdings = yes
		}
	}
}

fh_constitution_manila_conquest_holdings = {
	# JE04 saves its opening province scopes in this country variable list.
	# A snapshot stays authoritative after the journal closes.
	trigger_if = {
		limit = { has_variable_list = fh_manila_starting_provinces }
		fh_constitution_manila_recorded_balance > 0
		OR = {
			p:xD2CA4A ?= { owner ?= prev }
			NOT = {
				any_in_list = {
					variable = fh_manila_starting_provinces
					p:xD2CA4A ?= this
				}
			}
		}
	}
	trigger_else = {
		p:xD2CA4A ?= { owner ?= prev }
		fh_constitution_manila_ordinary_balance > 0
	}
}

fh_constitution_southern_support = {
	OR = {

	}
