fh_is_tungning_founder = {
	global_var:fh_tungning_founder ?= this
	is_country_alive = yes
}

fh_mainland_claim_unsettled = {
	fh_is_tungning_founder = yes
	has_journal_entry = fh_je_mainland_claim
	NOT = { has_variable = fh_mainland_settled }
}

fh_mainland_debate_active = {
	fh_mainland_claim_unsettled = yes
	NOR = {
		has_variable = fh_strait_offer_pending
		has_variable = fh_strait_ratification_in_progress
		has_variable = fh_taiping_fork_pending
		has_variable = fh_mainland_tone_pending
	}
}

	OR = {
		fh_owns_mainland_foothold = { REGION = STATE_FUJIAN }
		fh_owns_mainland_foothold = { REGION = STATE_GUANGDONG }
		fh_owns_mainland_foothold = { REGION = STATE_ZHEJIANG }
	}
}

	ig:ig_armed_forces ?= { is_in_government = yes }
	fh_mainland_officers_mainland = yes
	scaled_debt < 0.75
	in_default = no
}
